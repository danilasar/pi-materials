    \begin{enumerate}
        \item $\int x^n d x = \frac{x^{n+1}}{n+1} + C (n \neq -1)$
        \item $\int \frac 1 x dx = \ln | x | + C$
        \item $\int a^x d x = \frac{a^x}{\ln a} + C$
        \item $\int e^x d x = e^x + C$
        \item $\int \sin x d x = - \cos x + C$
        \item $\int \cos x d x = \sin x + C$
        \item $\int \frac{dx}{\sin^2 x} = - ctg \, x + C$
        \item $\int \frac{dx}{\cos^2 x} = tg \, x + C$
        \item $\int \frac{dx}{a^2 + x^2} = \frac 1 a arctg \, \frac x a + C (a \neq 1)$
        \item $\int \frac{dx}{\sqrt{a^2 - x^2}} = \arcsin \frac x a + C$
        \item $\int \frac{dx}{a^2 - x^2} = \frac 1 {2a} \ln \bigg| \frac{a - x}{a+ x} \bigg| + C$
        \item $\int \frac{xdx}{a^2 \pm x^2} = \pm \frac 1 2 \ln | a^2 \pm x^2 | + C$
        \item $\int \frac{dx}{\sqrt{a^2 \pm x^2}} = \pm sqrt{a^2 \pm x^2} + C$
        \item $\int \frac{dx}{\sqrt{x^2 \pm a^2}} = \ln | x + \sqrt{x^2 \pm a^2} | + C (a > 0)$
    \end{enumerate}
